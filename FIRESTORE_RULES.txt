rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isAuthor() {
      return request.auth != null
        && request.auth.uid == "KadiBgAh0sWJV1WvYMHOaWlFMn33";
    }

    function validLikeUpdate() {
      return request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likes'])
        && request.resource.data.likes is int
        && request.resource.data.likes >= 0
        && (
          request.resource.data.likes == resource.data.get('likes', 0) + 1
          || request.resource.data.likes == resource.data.get('likes', 0) - 1
        );
    }

    function validViewUpdate() {
      return request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['views'])
        && request.resource.data.views is int
        && request.resource.data.views >= 1
        && request.resource.data.views == resource.data.get('views', 0) + 1;
    }

    match /site/{document} {
      allow read: if true;
      allow create, update, delete: if isAuthor();
    }

    match /articles/{articleId} {
      allow read: if resource.data.status == "published" || isAuthor();
      allow create, delete: if isAuthor();
      allow update: if isAuthor() || validLikeUpdate() || validViewUpdate();
    }

    match /videos/{videoId} {
      allow read: if true;
      allow create, update, delete: if isAuthor();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
